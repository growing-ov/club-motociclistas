// Creacion de las colecciones del Club de Motociclistas en MongoDB
// Ejecutar en mongosh (o MongoDB Compass) con la base de datos del proyecto.
// use club_motociclistas

// ---------------------------------------------------------------- MOTOCICLISTAS
db.motociclistas.drop();

db.createCollection("motociclistas", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["nombre", "apellido", "correo", "fechaNacimiento", "activo"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador del motociclista" },
                nombre: { bsonType: "string", description: "Nombre del motociclista" },
                apellido: { bsonType: "string", description: "Apellido del motociclista" },
                correo: { bsonType: "string", description: "Correo electronico del miembro" },
                telefono: { bsonType: "string", description: "Telefono de contacto" },
                fechaNacimiento: { bsonType: "date", description: "Fecha de nacimiento" },
                activo: { bsonType: "bool", description: "Si el miembro sigue activo en el club" },
                tipoLicencia: { bsonType: "string", description: "Categoria de la licencia de moto" },
                fotoUrl: { bsonType: "string", description: "URL de la foto de perfil" }
            }
        }
    }
})

// -------------------------------------------------------------------- DOCUMENTOS
db.documentos.drop();

db.createCollection("documentos", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["nombreDocumento", "numeroDocumento", "fechaEmision", "fechaVencimiento", "idMotociclista"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador del documento" },
                nombreDocumento: { bsonType: "string", description: "Tipo de documento" },
                numeroDocumento: { bsonType: "string", description: "Numero del documento" },
                fechaEmision: { bsonType: "date", description: "Fecha de emision" },
                fechaVencimiento: { bsonType: "date", description: "Fecha de vencimiento" },
                archivoUrl: { bsonType: "string", description: "URL del archivo del documento" },
                idMotociclista: { bsonType: "objectId", description: "Socio dueño del documento" }
            }
        }
    }
})

// ------------------------------------------------------------------------ MOTOS
db.motos.drop();

db.createCollection("motos", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["placa", "marca", "modelo", "anio", "tipoMoto", "idMotociclista"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador de la moto" },
                placa: { bsonType: "string", description: "Placa de la moto" },
                marca: { bsonType: "string", description: "Marca de la moto" },
                modelo: { bsonType: "string", description: "Modelo de la moto" },
                anio: { bsonType: "int", description: "Anio del modelo", minimum: 1990, maximum: 2026 },
                tipoMoto: { bsonType: "string", description: "Tipo de moto" },
                cilindrada: { bsonType: "int", description: "Cilindrada en centimetros cubicos", minimum: 50, maximum: 2500 },
                color: { bsonType: "string", description: "Color de la moto" },
                fotoUrl: { bsonType: "string", description: "URL de la foto de la moto" },
                idMotociclista: { bsonType: "objectId", description: "Referencia al dueno en motociclistas" }
            }
        }
    }
})

// ------------------------------------------------------------------ MANTENIMIENTOS
db.mantenimientos.drop();

db.createCollection("mantenimientos", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["idMoto", "fecha", "tipo", "descripcion", "costo", "kilometraje"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador del mantenimiento" },
                idMoto: { bsonType: "objectId", description: "Referencia a la moto" },
                fecha: { bsonType: "date", description: "Fecha del mantenimiento" },
                tipo: { bsonType: "string", description: "Tipo de mantenimiento" },
                descripcion: { bsonType: "string", description: "Que se le hizo a la moto" },
                costo: { bsonType: "double", description: "Costo del mantenimiento", minimum: 0 },
                kilometraje: { bsonType: "int", description: "Kilometraje al momento del servicio", minimum: 0 },
                realizado: { bsonType: "bool", description: "Si el mantenimiento ya se hizo" }
            }
        }
    }
})

// --------------------------------------------------------------------- ACCIDENTES
db.accidentes.drop();

db.createCollection("accidentes", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["idMoto", "fecha", "descripcion", "ubicacion", "gravedad"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador del accidente" },
                idMoto: { bsonType: "objectId", description: "Referencia a la moto" },
                fecha: { bsonType: "date", description: "Fecha del accidente" },
                descripcion: { bsonType: "string", description: "Como succedio" },
                ubicacion: { bsonType: "string", description: "Lugar del accidente" },
                gravedad: { bsonType: "string", description: "Nivel de gravedad", enum: ["leve", "moderado", "grave"] },
                costoEstimado: { bsonType: "double", description: "Costo estimado de la reparacion", minimum: 0 }
            }
        }
    }
})

// ------------------------------------------------------------------------ RODADAS
db.rodadas.drop();

db.createCollection("rodadas", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["nombre", "fechaSalida", "puntoEncuentro", "destino", "estado", "cupoMaximo"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador de la rodada" },
                nombre: { bsonType: "string", description: "Nombre de la rodada" },
                fechaSalida: { bsonType: "date", description: "Fecha y hora de salida" },
                puntoEncuentro: { bsonType: "string", description: "Lugar donde se reunen" },
                destino: { bsonType: "string", description: "Destino de la rodada" },
                estado: { bsonType: "string", description: "Estado actual de la rodada", enum: ["programada", "en_curso", "finalizada"] },
                cupoMaximo: { bsonType: "int", description: "Numero maximo de inscritos", minimum: 2, maximum: 80 },
                distanciaKm: { bsonType: "int", description: "Distancia aproximada en kilometros", minimum: 1, maximum: 2000 },
                costo: { bsonType: "double", description: "Costo de inscripcion", minimum: 0 }
            }
        }
    }
})

// -------------------------------------------------------------------- INSCRIPCIONES
db.inscripciones.drop();

db.createCollection("inscripciones", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["idMotociclista", "idRodada", "fechaInscripcion", "asistio"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador de la inscripcion" },
                idMotociclista: { bsonType: "objectId", description: "Referencia al motociclista inscrito" },
                idRodada: { bsonType: "objectId", description: "Referencia a la rodada" },
                fechaInscripcion: { bsonType: "date", description: "Fecha en que se inscribio" },
                asistio: { bsonType: "bool", description: "Si el motociclista asistio a la rodada" },
                observaciones: { bsonType: "string", description: "Notas de la inscripcion" }
            }
        }
    }
})

// -------------------------------------------------------------------------- GALERIA
db.galeria.drop();

db.createCollection("galeria", {
    validator: {
        $jsonSchema: {
            bsonType: "object",
            required: ["idRodada", "titulo", "urlFoto", "fechaSubida"],
            properties: {
                _id: { bsonType: "objectId", description: "Identificador de la foto" },
                idRodada: { bsonType: "objectId", description: "Rodada a la que pertenece la foto" },
                titulo: { bsonType: "string", description: "Titulo de la foto" },
                descripcion: { bsonType: "string", description: "Descripcion de la foto" },
                urlFoto: { bsonType: "string", description: "URL de la imagen" },
                fechaSubida: { bsonType: "date", description: "Fecha en que se agrego la foto" }
            }
        }
    }
})
